        // =====================================================
        // パネルをドラッグ
        // =====================================================
        function enableDrag(panel, handle) {

            let dragging = false;
            let dx = 0;
            let dy = 0;

            handle.css({
                cursor: 'move',
                'user-select': 'none'
            });

            handle.on('mousedown', function (e) {

                if (e.button !== 0) return;

                const rect =
                    panel[0].getBoundingClientRect();

                dragging = true;

                dx = e.clientX - rect.left;
                dy = e.clientY - rect.top;

                panel.css({
                    transform: 'none',
                    left: rect.left + 'px',
                    top: rect.top + 'px'
                });

                e.preventDefault();
            });

            $(document).on(
                'mousemove.MapTroopDrag',
                function (e) {

                    if (!dragging) return;

                    let left = e.clientX - dx;
                    let top = e.clientY - dy;

                    left = Math.max(
                        0,
                        Math.min(
                            left,
                            window.innerWidth -
                            panel[0].offsetWidth
                        )
                    );

                    top = Math.max(
                        0,
                        Math.min(
                            top,
                            window.innerHeight - 40
                        )
                    );

                    panel.css({
                        left: left + 'px',
                        top: top + 'px'
                    });
                }
            );

            $(document).on(
                'mouseup.MapTroopDrag',
                function () {

                    if (!dragging) return;

                    dragging = false;

                    // =========================================
                    // 自動保存モードの場合
                    // ドラッグ終了位置を自動保存
                    // =========================================

                        load(KEY.autoPos, '0') === '1'

                            panel[0]
                                .getBoundingClientRect();

                        save(
                            KEY.left,
                            Math.round(rect.left)
                        );

                        save(
                            KEY.top,
                            Math.round(rect.top)
                        );
                    }
                }
            );
        }

            function getPreviewFields(cardId) {
                if (!previewForm.length) return null;
                const fields = previewForm.find(':input[name]').filter(function () {
                    const el = $(this);
                    const type = (el.attr('type') || '').toLowerCase();
                    if (['submit', 'button', 'image', 'file', 'reset'].includes(type)) return false;
                    if ((type === 'checkbox' || type === 'radio') && !el.prop('checked')) return false;
                    return !el.prop('disabled');
                }).map(function () {
                    return {name: this.name, value: $(this).val() || ''};
                }).get();
                function put(name, value) {
                    for (let i = fields.length - 1; i >= 0; i--) {
                        if (fields[i].name === name) fields.splice(i, 1);
                    }
                    fields.push({name, value: String(value)});
                }
                // 他武将のラジオ選択値が残らないようにする
                put('unit_assign_card_id', cardId);
                put('village_x_value', x);
                put('village_y_value', y);
                put('radio_reserve_type', 0);
                put('btn_preview', 302);
                // 明示的な手動発動スキルは選択しない（パッシブは公式側で適用）
                fields.forEach(f => {
                    if (/^use_skill_id\[/.test(f.name)) f.value = '0';
                    if (/^friendly_army_.*_count$/.test(f.name)) f.value = '0';
                });
                return fields.filter(f => !/^(btn_send|radio_move_type)$/.test(f.name));
            }

            // 公式画面の数値がテキスト・画像属性・スクリプト内の
            // いずれに含まれる場合も検索する。数値を推測して表示しない。
            function extractWinRate(html) {
                const doc = new DOMParser().parseFromString(html, 'text/html');
                const visible = (doc.body?.textContent || '').replace(/\s+/g, ' ');
                const raw = String(html).replace(/&nbsp;|&#160;/gi, ' ');
                const sources = [visible, raw];
                const patterns = [
                    /勝\s*利\s*確\s*率[\s\S]{0,180}?([0-9]{1,3}(?:\.[0-9]+)?)\s*[%％]/i,
                    /(?:win[_-]?rate|victory[_-]?rate|winning[_-]?rate|probability)[^0-9]{0,90}([0-9]{1,3}(?:\.[0-9]+)?)\s*[%％]?/i
                ];
                for (const str of sources) {
                    for (const pattern of patterns) {
                        const m = str.match(pattern);
                        if (m && Number(m[1]) <= 100) return m[1] + '%';
                    }
                }
                // 結果が画像化されている場合は、属性中に数値があるか確認
                for (const el of doc.querySelectorAll('img, [style], [data-rate]')) {
                    const attrs = ['alt', 'title', 'src', 'style', 'data-rate']
                        .map(k => el.getAttribute(k) || '').join(' ');
                    for (const pattern of patterns) {
                        const m = attrs.match(pattern);
                        if (m && Number(m[1]) <= 100) return m[1] + '%';
                    }
                }
                // 取得できないときの調査用（個人情報・HTML全文は出力しない）
                const title = doc.title || '';
                const hasRateLabel = /勝\s*利\s*確\s*率/.test(visible + raw);
                const hasSendButton = !!doc.querySelector('input[name="btn_send"], button[name="btn_send"]');
                console.info('[右クリ出兵v2.1.1] 確率解析結果', {
                    title, hasRateLabel, hasSendButton,
                    htmlLength: html.length,
                    imageCount: doc.images.length
                });
                return null;
            }

            function attachWinRateHover(table, cardId) {
                const img = table.find('img').filter(function () {
                    return $(this).closest('td').length > 0;
                }).first();
                if (!img.length) return;
                img.css('cursor', 'help');
                let tip = null;
                let timer = null;
                let currentToken = 0;
                img.on('mouseenter', function () {
                    if (moveSelect.val() !== '302') return;
                    currentToken = ++hoverToken;
                    const token = currentToken;
                    tip = $('<div>').css({
                        position: 'fixed', 'z-index': 111111115,
                        padding: '7px 12px', background: '#222', color: '#fff',
                        border: '1px solid #d9b44a', 'border-radius': '5px',
                        'font-size': '14px', 'font-weight': 'bold',
                        'pointer-events': 'none', 'white-space': 'nowrap'
                    }).text('勝利確率：取得中…').appendTo('body');
                    const r = this.getBoundingClientRect();
                    tip.css({left: Math.min(r.right + 8, window.innerWidth - 190) + 'px',
                        top: Math.max(0, r.top) + 'px'});
                    const key = [x, y, cardId].join(':');
                    if (previewCache.has(key)) {
                        tip.text(previewCache.get(key));
                        return;
                    }
                    timer = setTimeout(function () {
                        const fields = getPreviewFields(cardId);
                        if (!fields) {
                            if (token === hoverToken && tip) tip.text('勝利確率：フォーム取得不可');
                            return;
                        }
                        if (activePreview) activePreview.abort();
                        activePreview = $.ajax({
                            url: '/facility/castle_send_troop.php', type: 'POST',
                            data: $.param(fields), dataType: 'html'
                        }).done(function (html) {
                            const rate = extractWinRate(html);
                            const result = rate ? '勝利確率：' + rate : '勝利確率：解析不可（F12 Console確認）';
                            if (rate) previewCache.set(key, result);
                            if (token === hoverToken && tip) tip.text(result);
                        }).fail(function (_xhr, status) {
                            if (status !== 'abort' && token === hoverToken && tip)
                                tip.text('勝利確率：通信エラー');
                        }).always(function () { activePreview = null; });
                    }, 250);
                }).on('mouseleave', function () {
                    ++hoverToken;
                    clearTimeout(timer);
                    if (tip) tip.remove();
                    tip = null;
                    if (activePreview) activePreview.abort();
                });
            }

            function drawCards() {

                cardArea.empty();

                const hideLv0 =
                    lvSelect.val() === 'hide';

                const moveType =
                    parseInt(
                        moveSelect.val(),
                        10
                    );

                const buttonName =
                    MOVE_TYPES[moveType].name;

                source.each(function () {

                    const table =
                        $(this).clone();

                    // =========================================
                    // Lv0判定
                    // =========================================
                    if (
                        hideLv0 &&
                        table.html()
                            .indexOf(
                                '<strong>Lv</strong>0'
                            ) > -1
                    ) {
                        return;
                    }

                    table.css({
                        margin: 0
                    });

                    table
                        .find('th')
                        .remove();

                    table
                        .find('.left')
                        .remove();

                    // =========================================
                    // 武将選択radio
                    // =========================================
                    const radio =
                        table.find(
                            'input[name="unit_assign_card_id"]'
                        ).eq(0);

                    if (!radio.length) {
                        return;
                    }

                    const id =
                        radio.val();

                    // =========================================
                    // デッキ画面で非表示指定
                    // =========================================
                    if (
                        localStorage.getItem(
                            SERVER_NAME + '-' + id
                        ) === 'hide'
                    ) {
                        return;
                    }

                    // =========================================
                    // 2行目だけ残す
                    // =========================================
                    const targetTr =
                        table
                            .find('tr')
                            .eq(1)
                            .clone();

                    table
                        .find('tbody')
                        .empty()
                        .append(
                            targetTr
                        );

                    table
                        .find('input')
                        .remove();

                    // =========================================
                    // 出兵 / 強襲 / 援軍ボタン
                    // =========================================
                    const troopButton =
                        $('<button>')
                            .text(
                                buttonName
                            )
                            .css({
                                'font-weight':
                                    'bold'
                            })
                            .on(
                                'click',
                                function () {

                                    const currentType =
                                        parseInt(
                                            moveSelect.val(),
                                            10
                                        );

                                    const currentName =
                                        MOVE_TYPES[
                                            currentType
                                        ].name;

                                    const button =
                                        $(this);

                                    button
                                        .prop(
                                            'disabled',
                                            true
                                        )
                                        .text(
                                            currentName +
                                            '中...'
                                        );

                                    troop(
                                        x,
                                        y,
                                        id,
                                        currentType
                                    )
                                    .done(
                                        function () {

                                            tile
                                                .find(
                                                    '.tile-inner-rect'
                                                )
                                                .addClass(
                                                    'trooped'
                                                )
                                                .removeClass(
                                                    'clicked'
                                                )
                                                .css(
                                                    'fill',
                                                    '#00FF00'
                                                );

                                            $('#wrapTroopArea')
                                                .remove();

                                            $(document)
                                                .off(
                                                    '.MapTroopDrag'
                                                );
                                        }
                                    )
                                    .fail(
                                        function () {

                                            button
                                                .prop(
                                                    'disabled',
                                                    false
                                                )
                                                .text(
                                                    currentName
                                                );

                                            alert(
                                                currentName +
                                                'に失敗しました'
                                            );
                                        }
                                    );
                                }
                            );

                    table
                        .find('td')
                        .eq(0)
                        .append(
                            troopButton
                        );

                    attachWinRateHover(table, id);

                    cardArea.append(
                        table

        // マップ右クリック
        // =====================================================
        function contextFunction(e) {

            if (e.which !== 3) {
                return;
            }

            const x =
                e.currentTarget.dataset.x;

            const y =
                e.currentTarget.dataset.y;

            const tile =
                $(e.currentTarget);

            $('.clicked')
                .css(
                    'fill',
                    ''
                )
                .removeClass(
                    'clicked'
                );

            tile
                .find(
                    '.tile-inner-rect'
                )
                .addClass(
                    'clicked'
                )
                .css(
                    'fill',
                    '#FF00FF'
                );

            $('#wrapTroopArea')
                .remove();

            $(document)
                .off(
                    '.MapTroopDrag'
                );

            $.ajax({
                type: 'get',
                url:
                    '/facility/castle_send_troop.php'

            }).done(
                function (data) {

                    createPanel(
                        data,
                        x,
                        y,
                        tile
                    );
                }
            );

            e.preventDefault();
            return false;
        }
